import { useEffect, useState } from "react";
import { useAuth } from "../api/AuthContext";

export default function AdminDashboard() {
  const { apiClient } = useAuth();
  const [queueStatus, setQueueStatus] = useState(null);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState(null);
  const [autoRefresh, setAutoRefresh] = useState(true);

  useEffect(() => {
    const fetchStatus = async () => {
      try {
        const { data } = await apiClient.get("/admin/queue-status");
        setQueueStatus(data);
        setError(null);
      } catch (err) {
        setError(err.response?.data?.error || "Failed to fetch queue status");
      } finally {
        setLoading(false);
      }
    };

    fetchStatus();

    if (autoRefresh) {
      const interval = setInterval(fetchStatus, 5000); // Refresh every 5s
      return () => clearInterval(interval);
    }
  }, [apiClient, autoRefresh]);

  if (loading) return <div className="p-4">Loading queue status…</div>;
  if (error) return <div className="p-4 text-red-600">Error: {error}</div>;
  if (!queueStatus) return null;

  return (
    <div className="min-h-screen bg-ink-50 p-8">
      <div className="mx-auto max-w-6xl">
        <div className="mb-6 flex items-center justify-between">
          <h1 className="text-3xl font-bold text-ink-900">Queue Monitoring</h1>
          <label className="flex items-center gap-2 text-sm">
            <input
              type="checkbox"
              checked={autoRefresh}
              onChange={(e) => setAutoRefresh(e.target.checked)}
              className="h-4 w-4"
            />
            Auto-refresh (5s)
          </label>
        </div>

        <div className="grid grid-cols-1 gap-6 md:grid-cols-2">
          {Object.entries(queueStatus).map(([queueName, data]) => (
            <div key={queueName} className="rounded-lg border border-ink-200 bg-white p-6 shadow-sm">
              <h2 className="mb-4 text-lg font-semibold text-ink-900">{queueName}</h2>

              <div className="space-y-3 text-sm">
                <div className="flex justify-between">
                  <span className="text-ink-600">Active:</span>
                  <span className="font-mono font-bold text-highlight">{data.counts.active}</span>
                </div>
                <div className="flex justify-between">
                  <span className="text-ink-600">Waiting:</span>
                  <span className="font-mono font-bold text-blue-600">{data.counts.waiting}</span>
                </div>
                <div className="flex justify-between">
                  <span className="text-ink-600">Completed:</span>
                  <span className="font-mono font-bold text-green-600">{data.counts.completed}</span>
                </div>
                <div className="flex justify-between">
                  <span className="text-ink-600">Failed:</span>
                  <span className="font-mono font-bold text-red-600">{data.counts.failed}</span>
                </div>
                <div className="flex justify-between">
                  <span className="text-ink-600">Delayed:</span>
                  <span className="font-mono font-bold text-yellow-600">{data.counts.delayed}</span>
                </div>
              </div>

              {data.recentFailures.length > 0 && (
                <div className="mt-4 border-t border-ink-100 pt-4">
                  <h3 className="mb-2 text-xs font-semibold text-red-700">Recent Failures</h3>
                  <div className="space-y-2">
                    {data.recentFailures.map((failure) => (
                      <div key={failure.id} className="rounded bg-red-50 p-2">
                        <div className="text-xs font-mono text-ink-600">{failure.id}</div>
                        <div className="text-xs text-red-600">{failure.failedReason}</div>
                        <div className="text-xs text-ink-400">Attempt #{failure.attempts}</div>
                      </div>
                    ))}
                  </div>
                </div>
              )}
            </div>
          ))}
        </div>

        <div className="mt-8 rounded-lg border border-ink-200 bg-white p-6 shadow-sm">
          <h2 className="mb-4 text-lg font-semibold text-ink-900">Totals</h2>
          <div className="grid grid-cols-2 gap-4 md:grid-cols-5">
            {Object.entries(queueStatus).map(([queueName, data]) => (
              <div key={queueName} className="text-center">
                <div className="text-2xl font-bold text-highlight">{data.counts.completed}</div>
                <div className="text-xs text-ink-600">{queueName} completed</div>
              </div>
            ))}
          </div>
        </div>

        <div className="mt-4 text-xs text-ink-500">
          Last updated: {new Date().toLocaleTimeString()}
        </div>
      </div>
    </div>
  );
}
